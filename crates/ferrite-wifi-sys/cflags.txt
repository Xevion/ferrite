-Ilinux/include/uapi
-Ilinux/generated/include/generated/uapi
-include linux/compiler-version.h
-include linux/kconfig.h
-include linux/compiler_types.h
-std=gnu11
-fshort-wchar
-funsigned-char
-fno-common

-fno-delete-null-pointer-checks
-fno-strict-overflow
-fstrict-flex-arrays=3
-mgeneral-regs-only
-mno-red-zone
-fno-stack-protector
-O2
-fPIE
-fno-plt
-fdirect-access-external-data
-ffunction-sections
-fdata-sections
